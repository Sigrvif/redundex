# Redundex Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Redundex, a cross-platform (Windows/Linux) GUI tool that scans a folder for redundant files — exact duplicates, resizes, re-encodes, clips contained in longer videos, same audio tracks, same-content documents — and lets the user review and move them to a collection folder (never auto-delete).

**Architecture:** Qt-free Python core (`redundex/core/`) with per-type detectors coordinated by a single scan orchestrator using a `ProcessPoolExecutor`; fingerprints cached in a single-writer WAL SQLite store keyed by `(path, size, mtime)`. Thin PySide6 GUI layer over the core. Union-find merges pairwise matches into groups with redundancy sizes.

**Tech Stack:** Python ≥3.10, PySide6, Pillow, numpy, blake3, py7zr, pymupdf; external tools ffmpeg/ffprobe (video) and fpcalc (audio) detected at runtime, detectors degrade individually when missing.

**Spec:** `docs/specs/2026-08-22-redundancy-finder-design.md` (authoritative; two refinements locked here: (1) the containment prefilter uses an inverted index over 16-bit bands of *all sampled frames* in addition to the 16 fractional coarse hashes, because fractional coarse signatures do not align for a clip vs. its container; (2) tar listing hashes use `name|size|mtime` since tar carries no CRCs).

```python
import os

from .types import FileKind

_IMAGE = {"jpg", "jpeg", "png", "gif", "bmp", "webp", "tif", "tiff", "heic", "avif"}
_VIDEO = {"mp4", "mkv", "avi", "mov", "webm", "wmv", "m4v", "mpg", "mpeg", "ts", "flv"}
_AUDIO = {"mp3", "flac", "ogg", "oga", "wav", "m4a", "aac", "wma", "opus"}
_DOC = {"pdf", "txt", "md"}
_ARCHIVE = {"zip", "7z", "tar", "gz", "bz2", "xz"}

_SETS = (
    (_IMAGE, FileKind.IMAGE),
    (_VIDEO, FileKind.VIDEO),
    (_AUDIO, FileKind.AUDIO),
    (_DOC, FileKind.DOC),
    (_ARCHIVE, FileKind.ARCHIVE),
)


def classify(path: str) -> FileKind:
    ext = os.path.splitext(path)[1].lower().lstrip(".")
    for exts, kind in _SETS:
        if ext in exts:
            return kind
    return FileKind.OTHER
```

```python
import os
import stat

from .classify import classify
from .types import FileEntry

_CACHE_PREFIX = ".redundex.cache.sqlite"


def _normalize(path: str) -> str:
    if os.name == "nt" and len(path) > 240 and not path.startswith("\\\\?\\"):
        return "\\\\?\\" + os.path.abspath(path)
    return path


def walk_folder(root: str, follow_symlinks: bool = False):
    entries = []
    problems = []
    root = os.path.abspath(root)
    for dirpath, dirnames, filenames in os.walk(root, followlinks=follow_symlinks):
        if not follow_symlinks:
            dirnames[:] = [
                d for d in dirnames if not os.path.islink(os.path.join(dirpath, d))
            ]
        for name in filenames:
            p = os.path.join(dirpath, name)
            if name.startswith(_CACHE_PREFIX):
                continue
            if not follow_symlinks and os.path.islink(p):
                continue
            try:
                st = os.stat(p, follow_symlinks=follow_symlinks)
                if not stat.S_ISREG(st.st_mode):
                    continue
                entries.append(
                    FileEntry(
                        path=_normalize(p),
                        size=st.st_size,
                        mtime_ns=st.st_mtime_ns,
                        kind=classify(p),
                    )
                )
            except OSError as exc:
                problems.append({"path": p, "error": str(exc)})
    return entries, problems
```

```python
import concurrent.futures
import multiprocessing as mp
import os


class CancelledScan(Exception):
    pass


def run_jobs(fn, items, workers=None, cancel_event=None, progress_cb=None):
    results = []
    problems = []
    items = list(items)
    if not items:
        return results, problems
    n = workers or os.cpu_count() or 4
    ctx = mp.get_context("spawn")
    with concurrent.futures.ProcessPoolExecutor(max_workers=n, mp_context=ctx) as ex:
        futures = {ex.submit(fn, it): it for it in items}
        done = 0
        try:
            for fut in concurrent.futures.as_completed(futures):
                if cancel_event is not None and cancel_event.is_set():
                    raise CancelledScan()
                item = futures[fut]
                try:
                    results.append(fut.result())
                except Exception as exc:
                    problems.append({"item": item, "error": str(exc)})
                done += 1
                if progress_cb is not None:
                    progress_cb(done, len(items))
        except CancelledScan:
            for f in futures:
                f.cancel()
            raise
    return results, problems
```

```python
import os
import time

import pytest

from redundex.core.hashstore import CacheLockedError, HashStore
from redundex.core.types import FileKind
from redundex.core.walk import walk_folder


def _first_entry(root):
    entries, _ = walk_folder(root)
    return {e.path: e for e in entries}


def test_roundtrip_freshness_and_prune(make_tree, tmp_path_factory):
    root = make_tree({"a.jpg": b"12345", "b.jpg": b"678"})
    entries = _first_entry(root)
    cache = str(tmp_path_factory.mktemp("cache") / "c.sqlite")
    store = HashStore(root, cache_path=cache)
    assert store.load("fp_exact", entries) == {}
    store.save_files(entries)
    store.put(
        "fp_exact",
        [
            {"path": e.path, "file_hash": "h" + os.path.basename(e.path), "listing_hash": None}
            for e in entries.values()
        ],
    )
    fresh = store.load("fp_exact", entries)
    assert {p: fresh[p]["file_hash"] for p in entries} == {
        str(root / "a.jpg"): "ha.jpg",
        str(root / "b.jpg"): "hb.jpg",
    }

    os.utime(str(root / "a.jpg"), ns=(time.time_ns() + 10**9,) * 2)
    stale_entries = _first_entry(root)
    assert set(store.load("fp_exact", stale_entries)) == {str(root / "b.jpg")}

    (root / "b.jpg").unlink()
    store.prune({str(root / "a.jpg")})
    store.close()
    store = HashStore(root, cache_path=cache)
    assert store.load("fp_exact", stale_entries) == {}
    remaining = [r[0] for r in store.db.execute("SELECT path FROM files")]
    assert remaining == [str(root / "a.jpg")]
    store.close()


def test_lock_prevents_second_instance(make_tree, tmp_path):
    root = make_tree({"a.jpg": b"x"})
    cache = str(tmp_path / "c.sqlite")
    store = HashStore(root, cache_path=cache)
    with pytest.raises(CacheLockedError):
        HashStore(root, cache_path=cache)
    store.close()
    HashStore(root, cache_path=cache).close()


def test_stale_lock_is_taken_over(make_tree, tmp_path):
    import time as _time

    root = make_tree({"a.jpg": b"x"})
    cache = str(tmp_path / "c.sqlite")
    store = HashStore(root, cache_path=cache)
    old = _time.time() - 48 * 3600
    os.utime(store.lock_path, (old, old))
    store2 = HashStore(root, cache_path=cache)
    store2.close()
    store.close()


def test_default_cache_location(make_tree):

    store = HashStore(root)
    assert store.path.name == ".redundex.cache.sqlite"
    assert store.path.parent == root.resolve()
    store.close()
```

```python
import os
import sqlite3
import time
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS files(
    path TEXT PRIMARY KEY, size INTEGER, mtime_ns INTEGER, kind TEXT);
CREATE TABLE IF NOT EXISTS fp_exact(
    path TEXT PRIMARY KEY, file_hash TEXT, listing_hash TEXT);
CREATE TABLE IF NOT EXISTS fp_image(
    path TEXT PRIMARY KEY, phash INTEGER, dhash INTEGER, width INTEGER, height INTEGER);
CREATE TABLE IF NOT EXISTS fp_video(
    path TEXT PRIMARY KEY, duration REAL, frame_hashes BLOB, coarse BLOB,

    path TEXT PRIMARY KEY, raw BLOB, duration REAL);
CREATE TABLE IF NOT EXISTS fp_doc(
    path TEXT PRIMARY KEY, text_hash TEXT, page1_phash INTEGER);
"""

_TABLES = frozenset(
    {"files", "fp_exact", "fp_image", "fp_video", "fp_audio", "fp_doc"}
)


class CacheLockedError(RuntimeError):
    pass


class HashStore:
    def __init__(self, root: str, cache_path: str | None = None):
        p = Path(cache_path) if cache_path else Path(root) / ".redundex.cache.sqlite"
        self.path = p
        self.lock_path = p.with_name(p.name + ".lock")
        self._acquire_lock()
        self.db = sqlite3.connect(str(p))
        self.db.execute("PRAGMA journal_mode=WAL")
        self.db.executescript(SCHEMA)
        self.db.commit()

    def _acquire_lock(self, stale_after=24 * 3600):
        for _ in range(2):
            try:
                fd = os.open(
                    str(self.lock_path), os.O_CREAT | os.O_EXCL | os.O_WRONLY
                )
            except FileExistsError:
                try:
                    age = time.time() - os.path.getmtime(str(self.lock_path))
                except OSError:
                    age = 0
                if age > stale_after:
                    try:
                        self.lock_path.unlink()
                    except FileNotFoundError:
                        pass
                    continue
                raise CacheLockedError(
                    f"cache is locked by another instance: {self.lock_path}"
                )
            os.write(fd, str(os.getpid()).encode())
            os.close(fd)
            return

    def _table(self, table):
        if table not in _TABLES:
            raise ValueError(f"unknown table: {table}")
        return table

    def save_files(self, entries):
        rows = [
            (e.path, e.size, e.mtime_ns, e.kind.value) for e in entries.values()
        ]
        self.db.executemany(
            "INSERT OR REPLACE INTO files(path, size, mtime_ns, kind) VALUES(?,?,?,?)",
            rows,
        )
        self.db.commit()

    def load(self, table, entries):
        table = self._table(table)
        fresh = {}
        cur = self.db.execute(
            f"SELECT f.size AS _size, f.mtime_ns AS _mtime, t.* "
            f"FROM {table} t JOIN files f ON f.path = t.path"
        )
        cols = [d[0] for d in cur.description]
        for row in cur.fetchall():
            d = dict(zip(cols, row))
            e = entries.get(d["path"])
            if e is None:
                continue
            if e.size == d["_size"] and e.mtime_ns == d["_mtime"]:
                d.pop("_size"), d.pop("_mtime")
                fresh[d["path"]] = d
        return fresh

    def put(self, table, rows):
        table = self._table(table)
        if not rows:
            return
        cols = list(rows[0].keys())
        sql = (
            f"INSERT OR REPLACE INTO {table}({','.join(cols)}) "
            f"VALUES({','.join('?' * len(cols))})"
        )
        self.db.executemany(sql, [tuple(r[c] for c in cols) for r in rows])
        self.db.commit()

    def prune(self, alive):
        all_paths = [r[0] for r in self.db.execute("SELECT path FROM files")]
        vanished = [p for p in all_paths if p not in alive]
        for i in range(0, len(vanished), 500):
            chunk = vanished[i : i + 500]
            marks = ",".join("?" * len(chunk))
            self.db.execute(f"DELETE FROM files WHERE path IN ({marks})", chunk)
            for table in ("fp_exact", "fp_image", "fp_video", "fp_audio", "fp_doc"):
                self.db.execute(f"DELETE FROM {self._table(table)} WHERE path IN ({marks})", chunk)
        self.db.commit()

    def close(self):
        try:
            self.db.commit()
            self.db.close()
        finally:
            try:
                self.lock_path.unlink()
            except FileNotFoundError:
                pass

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()
```

```python
import os
import tarfile
import zipfile

import blake3

from ..types import Match, MatchReason


def blake3_file(path: str, chunk_size: int = 1 << 20) -> str:
    h = blake3.blake3()
    with open(path, "rb") as f:
        while True:
            b = f.read(chunk_size)
            if not b:
                break
            h.update(b)
    return h.hexdigest()


def archive_listing_hash(path: str):
    ext = os.path.splitext(path)[1].lower()
    members = []
    try:
        if ext == ".zip":
            with zipfile.ZipFile(path) as z:
                for i in z.infolist():
                    members.append((i.filename, i.file_size, i.CRC))
        elif ext in (".tar", ".gz", ".bz2", ".xz"):
            with tarfile.open(path, "r:*") as t:
                for m in t.getmembers():
                    members.append((m.name, m.size, m.mtime))
        elif ext == ".7z":
            try:
                import py7zr
            except ImportError:
                return None
            with py7zr.SevenZipFile(path) as z:
                for i in z.list():
                    members.append((i.filename, i.uncompressed, i.crc32 or 0))
        else:
            return None
    except Exception:
        return None
    members.sort()
    joined = "\x00".join(f"{n}|{s}|{c}" for n, s, c in members)
    return blake3.blake3(joined.encode("utf-8", "replace")).hexdigest()


def hash_job(path):
    return path, blake3_file(path), archive_listing_hash(path)


def match_from_fingerprints(fps):
    by_file_hash = {}
    by_listing = {}
    for path, (fh, listing) in fps.items():
        by_file_hash.setdefault(fh, []).append(path)
        if listing:
            by_listing.setdefault(listing, []).append(path)
    matches = []
    for group in by_file_hash.values():
        if len(group) > 1:
            for other in group[1:]:
                matches.append(
                    Match(
                        a=group[0],
                        b=other,
                        reason=MatchReason.EXACT,
                        score=1.0,
                        detail="identical content",
                    )
                )
    exact_pairs = {(m.a, m.b) for m in matches}
    for group in by_listing.values():
        if len(group) > 1:
            for i in range(len(group)):
                for j in range(i + 1, len(group)):
                    a, b = sorted((group[i], group[j]))
                    if (a, b) in exact_pairs or fps[a][0] == fps[b][0]:
                        continue
                    matches.append(
                        Match(
                            a=a,
                            b=b,
                            reason=MatchReason.EXACT,
                            score=1.0,
                            detail="same archive contents",
                        )
                    )
    return matches
```

```python
from .types import Group


class UnionFind:
    def __init__(self, items):
        self.parent = {i: i for i in items}

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra != rb:
            self.parent[rb] = ra


def build_groups(entries, matches, quality=None):
    quality = quality or {}
    uf = UnionFind(entries.keys())
    for m in matches:
        if m.a in entries and m.b in entries:
            uf.union(m.a, m.b)
    clusters = {}
    for path in entries:
        clusters.setdefault(uf.find(path), []).append(path)
    groups = []
    for members in clusters.values():
        if len(members) < 2:
            continue
        members.sort(
            key=lambda p: (quality.get(p, entries[p].size), entries[p].mtime_ns),
            reverse=True,
        )
        member_set = set(members)
        gmatches = [m for m in matches if m.a in member_set and m.b in member_set]
        best_size = entries[members[0]].size
        redundancy = sum(entries[p].size for p in members) - best_size
        groups.append(
            Group(
                members=[entries[p] for p in members],
                matches=gmatches,
                redundancy=redundancy,
            )
        )
    groups.sort(key=lambda g: g.redundancy, reverse=True)
    return groups
```

```python
import multiprocessing as mp
import time

from . import match as matching
from .detectors import exact
from .hashstore import HashStore
from .pool import CancelledScan, run_jobs
from .types import FileKind, ScanConfig, ScanResult
from .walk import walk_folder


class Scan:
    def __init__(self, root, config=None, progress_cb=None, cancel_event=None):
        self.root = str(root)
        self.config = config or ScanConfig()
        self.progress_cb = progress_cb
        self.cancel_event = cancel_event if cancel_event is not None else mp.Event()

    def _progress(self, stage, done, total):
        if self.progress_cb is not None:
            self.progress_cb(stage, done, total)

    def _cancelled(self):
        return self.cancel_event.is_set()

    def run(self) -> ScanResult:
        result = ScanResult()
        cfg = self.config
        try:
            store = HashStore(self.root, cfg.cache_path)
        except Exception as exc:
            result.problems.append({"path": "cache", "error": str(exc)})
            return result
        try:
            self._progress("walk", 0, 1)
            entries_list, problems = walk_folder(
                self.root, follow_symlinks=cfg.follow_symlinks
            )
            result.problems.extend(problems)
            entries = {e.path: e for e in entries_list}
            store.save_files(entries)
            store.prune(set(entries))
            self._progress("walk", 1, 1)
            if self._cancelled():
                result.cancelled = True
                return result

            matches = []
            quality = {}
            if cfg.exact:
                self._exact_stage(store, entries, matches, result)
            if self._cancelled():
                result.cancelled = True
                return result

            self._progress("group", 0, 1)
            result.groups = matching.build_groups(entries, matches, quality)
            result.stats["groups"] = len(result.groups)
            self._progress("group", 1, 1)
            return result
        except CancelledScan:
            result.cancelled = True
            return result
        finally:
            store.close()

    def _exact_stage(self, store, entries, matches, result):
        cfg = self.config
        t0 = time.monotonic()
        by_size = {}
        for e in entries.values():
            by_size.setdefault(e.size, []).append(e)
        candidates = {
            e.path
            for group in by_size.values()
            if len(group) > 1
            for e in group
        }
        candidates.update(
            e.path for e in entries.values() if e.kind is FileKind.ARCHIVE
        )
        cached = store.load("fp_exact", entries)
        missing = [p for p in candidates if p not in cached]
        self._progress("exact", 0, max(len(missing), 1))
        rows, problems = run_jobs(
            exact.hash_job,
            missing,
            workers=cfg.workers,
            cancel_event=self.cancel_event,
            progress_cb=lambda d, t: self._progress("exact", d, t),
        )
        result.problems.extend(problems)
        store.put(
            "fp_exact",
            [
                {"path": p, "file_hash": fh, "listing_hash": listing}
                for p, fh, listing in rows
            ],
        )
        fps = {p: (r["file_hash"], r["listing_hash"]) for p, r in cached.items()}
        fps.update({p: (fh, listing) for p, fh, listing in rows})
        stage_matches = exact.match_from_fingerprints(fps)
        matches.extend(stage_matches)
        result.stats["exact"] = {
            "candidates": len(candidates),
            "hashed": len(rows),
            "matches": len(stage_matches),
            "seconds": round(time.monotonic() - t0, 3),
        }
```

```python
def _format_size(n):
```

```python
import numpy
from PIL import Image

_SIZE = 32


def _dct_matrix(n):
    k = numpy.arange(n).reshape(n, 1).astype(numpy.float64)
    i = numpy.arange(n).reshape(1, n).astype(numpy.float64)
    m = numpy.cos(numpy.pi * (2 * k + 1) * i / (2 * n)) * numpy.sqrt(2.0 / n)
    m[0, :] /= numpy.sqrt(2.0)
    return m


_DCT = _dct_matrix(_SIZE)


def _pack_bits(bits):
    return int.from_bytes(numpy.packbits(bits, bitorder="big").tobytes(), "big")


def phash_gray(gray):
    d = _DCT @ gray @ _DCT.T
    low = d[:8, :8]
    return _pack_bits((low > numpy.median(low)).flatten())


def _box_downsample(gray, size):
    h, w = gray.shape
    bh, bw = h // size, w // size
    return gray[: bh * size, : bw * size].reshape(size, bh, size, bw).mean(axis=(1, 3))


def hash_gray_frame(gray):
    g = numpy.asarray(gray, dtype=numpy.float64)
    if g.shape != (_SIZE, _SIZE):
        g = _box_downsample(g, _SIZE)
    return phash_gray(g)


def phash_image(path):
    with Image.open(path) as im:
        gray = numpy.asarray(
            im.convert("L").resize((_SIZE, _SIZE), Image.LANCZOS), dtype=numpy.float64
        )
    return phash_gray(gray)


def dhash_image(path):
    with Image.open(path) as im:
        g = numpy.asarray(
            im.convert("L").resize((9, 8), Image.LANCZOS), dtype=numpy.float64
        )
    return _pack_bits((g[:, 1:] > g[:, :-1]).flatten())


def image_job(path):
    with Image.open(path) as im:
        w, h = im.size
        gray32 = numpy.asarray(
            im.convert("L").resize((_SIZE, _SIZE), Image.LANCZOS),
            dtype=numpy.float64,
        )
        g98 = numpy.asarray(
            im.convert("L").resize((9, 8), Image.LANCZOS), dtype=numpy.float64
        )
    dh = _pack_bits((g98[:, 1:] > g98[:, :-1]).flatten())
    return path, phash_gray(gray32), dh, w, h


def hamming(a, b):
    return (a ^ b).bit_count()


def to_signed64(v):
    return v - 2**64 if v >= 2**63 else v


def from_signed64(v):
    return v + 2**64 if v < 0 else v
```

SQLite INTEGER columns are signed 64-bit; hashes are unsigned 64-bit, so all persistence of hash values goes through `to_signed64`/`from_signed64` (XOR/`bit_count` on negative ints is wrong, so always convert back after loading). Later chunks (`fp_doc.page1_phash`, video frame hashes) must reuse these helpers.

```python
class _Node:
    __slots__ = ("item", "children")

    def __init__(self, item):
        self.item = item
        self.children = {}


class BKTree:
    def __init__(self, distance=None):
        if distance is None:
            from .imagehash import hamming

            distance = hamming
        self.distance = distance
        self.root = None

    def add(self, item):
        if self.root is None:
            self.root = _Node(item)
            return
        node = self.root
        while True:
            d = self.distance(item, node.item)
            if d == 0:
                return
            child = node.children.get(d)
            if child is None:
                node.children[d] = _Node(item)
                return
            node = child

    def query(self, target, radius):
        found = []
        if self.root is None:
            return found
        stack = [self.root]
        while stack:
            node = stack.pop()
            d = self.distance(target, node.item)
            if d <= radius:
                found.append((d, node.item))
            lo, hi = d - radius, d + radius
            for dist, child in node.children.items():
                if lo <= dist <= hi:
                    stack.append(child)
        return found
```

```python
from ..types import Match, MatchReason
from .bktree import BKTree
from .imagehash import hamming


def _pair_distance(a, b):
    return hamming(a[0], b[0])


def match_images(fps, threshold=8):
    phash_tree = BKTree(distance=_pair_distance)
    dhash_tree = BKTree(distance=_pair_distance)
    entries = [(ph, dh, path) for path, (ph, dh, _w, _h) in fps.items()]
    for ph, _dh, path in entries:
        phash_tree.add((ph, path))
    for _ph, dh, path in entries:
        dhash_tree.add((dh, path))
    best = {}
    for ph, dh, path in entries:
        for d, other in phash_tree.query((ph, path), threshold):
            if other[1] != path:
                pair = tuple(sorted((path, other[1])))
                best[pair] = min(best.get(pair, 65), d)
        for d, other in dhash_tree.query((dh, path), threshold):
            if other[1] != path:
                pair = tuple(sorted((path, other[1])))
                best[pair] = min(best.get(pair, 65), d)
    matches = []
    for (a, b), d in sorted(best.items()):
        matches.append(
            Match(
                a=a,
                b=b,
                reason=MatchReason.PERCEPTUAL,
                score=1.0 - d / 64.0,
                detail=f"hamming distance {d}/64",
            )
        )
    return matches
```

```python
import json
import os
import shutil
import time


def _unique_dest(dest):
    if not os.path.exists(dest):
        return dest
    base, ext = os.path.splitext(dest)
    i = 2
    while True:
        candidate = f"{base} ({i}){ext}"
        if not os.path.exists(candidate):
            return candidate
        i += 1


def _manifest_path(collection_dir):
    today = time.strftime("%Y%m%d")
    seq = 1
    while True:
        p = os.path.join(collection_dir, f"undo_{today}-{seq:03d}.json")
        if not os.path.exists(p):
            return p
        seq += 1


def latest_manifest(collection_dir):
    if not os.path.isdir(collection_dir):
        return None
    manifests = sorted(
        f
        for f in os.listdir(collection_dir)
        if f.startswith("undo_") and f.endswith(".json")
    )
    return os.path.join(collection_dir, manifests[-1]) if manifests else None


def move_batch(paths, collection_dir, root=None):
    os.makedirs(collection_dir, exist_ok=True)
    manifest_path = _manifest_path(collection_dir)
    moved = []
    failed = []
    for src in paths:
        try:
            rel = os.path.relpath(src, root) if root else os.path.basename(src)
            dest = os.path.join(collection_dir, rel)
            os.makedirs(os.path.dirname(dest), exist_ok=True)
            dest = _unique_dest(dest)
            shutil.move(src, dest)
            moved.append({"src": src, "dst": dest})
        except OSError as exc:
            failed.append({"path": src, "error": str(exc)})
    with open(manifest_path, "w", encoding="utf-8") as f:
        json.dump(
            {"created": time.strftime("%Y-%m-%dT%H:%M:%S"), "moves": moved},
            f,
            indent=2,
        )
    return manifest_path, moved, failed


def _restore_target(collection_dir, src):
    d = os.path.join(collection_dir, "restored")
    os.makedirs(d, exist_ok=True)
    dest = os.path.join(d, os.path.basename(src))
    if not os.path.exists(dest):
        return dest
    base, ext = os.path.splitext(dest)
    i = 1
    while True:
        candidate = f"{base} (restored-{i}){ext}"
        if not os.path.exists(candidate):
            return candidate
        i += 1


def undo_last(collection_dir):
    manifest_path = latest_manifest(collection_dir)
    if manifest_path is None:
        return {"restored": 0, "remaining": 0, "manifest": None}
    with open(manifest_path, encoding="utf-8") as f:
        data = json.load(f)
    remaining = []
    restored = 0
    for move in data.get("moves", []):
        src, dst = move["src"], move["dst"]
        try:
            if not os.path.exists(dst):
                continue
            target = src
            if os.path.exists(target):
                target = _restore_target(collection_dir, src)
            os.makedirs(os.path.dirname(target), exist_ok=True)
            shutil.move(dst, target)
            restored += 1
        except OSError:
            remaining.append(move)
    if remaining:
        data["moves"] = remaining
        with open(manifest_path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2)
    else:
        os.remove(manifest_path)
    return {
        "restored": restored,
        "remaining": len(remaining),
        "manifest": manifest_path,
    }
```
